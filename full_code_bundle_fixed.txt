import { nice } from '../utils/nice';

function strengthWord(pct) {
  const a = Math.abs(pct);
  if (a > 300) return pct >= 0 ? 'by far the strongest driver' : 'the strongest blocker';
  if (a > 100) return pct >= 0 ? 'a very strong driver' : 'a strong blocker';
  if (a > 30) return pct >= 0 ? 'a moderate driver' : 'a moderate blocker';
  return pct >= 0 ? 'a mild positive signal' : 'a mild negative signal';
}

function interpretation(feature, pct, p) {
  const name = nice(feature);
  if (p > 0.05) {
    return `${name} shows a ${pct >= 0 ? 'positive' : 'negative'} association, but it isn't statistically reliable (p > 0.05) — not a basis for action on its own.`;
  }
  if (pct >= 0) {
    return `People with ${name.toLowerCase()} are substantially more likely to be nominated than comparable peers. This is ${strengthWord(pct)} in this market.`;
  }
  return `${name} works against the move — people with it are less likely to progress in the near term. It is ${strengthWord(pct)} here.`;
}

function tag(pct, p) {
  if (p > 0.05) return { text: 'LOW CONFIDENCE', bg: 'var(--amber-tint)', col: 'var(--amber)' };
  if (pct >= 300) return { text: 'STRONGEST', bg: 'var(--status-red-tint)', col: 'var(--status-red)' };
  if (pct >= 0) return { text: 'POSITIVE', bg: 'var(--green-tint)', col: 'var(--green)' };
  return { text: 'BLOCKER', bg: 'var(--status-red-tint)', col: 'var(--status-red)' };
}

function directionWord(pct) {
  const a = Math.abs(pct);
  if (pct >= 0) return a > 300 ? 'strongest ▲' : a > 100 ? 'much higher ▲' : a > 30 ? 'higher ▲' : 'slightly ▲';
  return a > 100 ? 'much lower ▼' : a > 30 ? 'lower ▼' : 'slightly ▼';
}

function Card({ r, maxAbs }) {
  const t = tag(r.pct, r.p);
  const faded = r.p > 0.05;
  const barCol = faded ? '#c7c3ba' : (r.pct >= 0 ? 'var(--green)' : 'var(--status-red)');
  const strength = Math.min(100, (Math.abs(r.pct) / maxAbs) * 100);
  const confidence = r.p > 0.05 ? 'Low confidence' : (r.p < 0.01 ? 'High confidence' : 'Moderate confidence');

  return (
    <div className="card-rich" style={{ background: 'var(--surface-1)', border: '1px solid var(--border)', borderRadius: 12, padding: '18px 20px' }}>
      <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
        <span style={{ fontSize: 14, fontWeight: 600 }}>{nice(r.feature)}</span>
        <span style={{ fontSize: 9, letterSpacing: '.06em', padding: '3px 8px', borderRadius: 5, fontWeight: 700, background: t.bg, color: t.col }}>{t.text}</span>
      </div>
      <p style={{ fontSize: 32, fontWeight: 700, textAlign: 'center', margin: '14px 0 2px', color: faded ? 'var(--text-muted)' : (r.pct >= 0 ? 'var(--green)' : 'var(--status-red)') }}>
        {directionWord(r.pct)}
      </p>
      <p style={{ fontSize: 11, color: 'var(--text-muted)', textAlign: 'center', marginBottom: 16 }}>relative strength in this market</p>

      <div style={{ height: 6, background: 'var(--surface-sunk)', borderRadius: 3, marginBottom: 16, overflow: 'hidden' }}>
        <div style={{ height: '100%', width: `${strength}%`, background: barCol, borderRadius: 3 }} />
      </div>

      <div style={{ borderTop: '1px solid var(--border)', paddingTop: 12 }}>
        <p style={{ fontSize: 9, letterSpacing: '.06em', color: 'var(--text-faint)', textTransform: 'uppercase', margin: '0 0 6px' }}>✦ Interpretation</p>
        <p style={{ fontSize: 12, lineHeight: 1.55, color: 'var(--text-secondary)', margin: 0 }}>{interpretation(r.feature, r.pct, r.p)}</p>
        <p style={{ fontSize: 10, color: 'var(--text-faint)', textAlign: 'right', margin: '8px 0 0' }}>{confidence}</p>
      </div>
    </div>
  );
}

export default function OverviewView({ driverRows, market, block }) {
  const sorted = [...driverRows].sort((a, b) => Math.abs(b.pct) - Math.abs(a.pct));
  const top4 = sorted.slice(0, 4);
  const maxAbs = Math.max(...sorted.map(r => Math.abs(r.pct)), 1);
  const goal = block.replace('_vs_Talent', '');

  const strongest = sorted[0];
  const blocker = sorted.find(r => r.pct < 0 && r.p <= 0.05);
  const riskyCount = driverRows.filter(r => r.p > 0.05).length;

  const summary = `In ${market}, ${nice(strongest.feature).toLowerCase()} is ${strengthWord(strongest.pct)} into ${goal} — people with it are far more likely to be nominated.` +
    (blocker ? ` By contrast, ${nice(blocker.feature).toLowerCase()} works against the near-term move.` : '') +
    (riskyCount > 0 ? ` ${riskyCount} driver${riskyCount > 1 ? 's are' : ' is'} not statistically reliable and should be read with caution.` : '');

  return (
    <div>
      <div className="card-rich" style={{ background: 'var(--surface-1)', border: '1px solid var(--border)', borderRadius: 12, padding: '18px 20px', marginBottom: 18 }}>

        <p style={{ fontSize: 13.5, lineHeight: 1.65, color: 'var(--text-secondary)', margin: 0 }}>
          <b style={{ color: 'var(--text-primary)' }}>Overview:</b> {summary}
        </p>
      </div>

      <div style={{ display: 'grid', gridTemplateColumns: '1fr 1fr', gap: 14 }}>
        {top4.map(r => <Card key={r.feature} r={r} maxAbs={maxAbs} />)}
      </div>
    </div>
  );
}

function plainLabel(pct) {
  const a = Math.abs(pct);
  if (pct >= 0) {
    if (a > 300) return 'strongest ▲';
    if (a > 100) return 'much higher ▲';
    if (a > 30) return 'higher ▲';
    return 'slightly ▲';
  } else {
    if (a > 100) return 'much lower ▼';
    if (a > 30) return 'lower ▼';
    return 'slightly ▼';
  }
}

function impactSentence(feature, pct, p) {
  if (p > 0.05) return `${nice(feature)} shows a ${pct >= 0 ? 'positive' : 'negative'} association, but it's not statistically reliable — treat with caution.`;
  if (pct >= 0) return `People with ${nice(feature).toLowerCase()} are more likely to progress. ${Math.abs(pct) > 300 ? 'This is the strongest lever here.' : ''}`;
  return `${nice(feature)} works against the near-term move — people with it are less likely to progress.`;
}

function kpiFact(label, market, allMarketKpis) {
  const val = (m, key) => allMarketKpis[m]?.[key] ?? 0;
  const avg = (key) => Object.values(allMarketKpis).reduce((s, k) => s + k[key], 0) / Object.keys(allMarketKpis).length;
  if (label === 'Headcount') {
    const biggest = Object.keys(allMarketKpis).reduce((a, b) => allMarketKpis[a].n > allMarketKpis[b].n ? a : b);
    return market === biggest ? `${market} is the largest population of the three markets.` : `${market} is smaller than ${biggest}, the largest market.`;
  }
  const key = label === 'Talent' ? 'talentRate' : label === 'Accelerated' ? 'acceleratedRate' : 'topRate';
  const here = val(market, key);
  const diff = here - avg(key);
  const dir = diff >= 0 ? 'above' : 'below';
  const rank = Object.keys(allMarketKpis).sort((a, b) => val(b, key) - val(a, key)).indexOf(market) + 1;
  return `At ${here.toFixed(1)}%, ${market} is ${Math.abs(diff).toFixed(1)} pts ${dir} the cross-market average — rank ${rank} of ${Object.keys(allMarketKpis).length}.`;
}

function Bar({ row, min, max, selected, onClick, onHover, onLeave }) {
  const { feature, pct, p } = row;
  const span = max - min || 1;
  const zero = ((0 - min) / span) * 100;
  const size = (Math.abs(pct) / span) * 100;
  const left = pct >= 0 ? zero : zero - size;
  const faded = p > 0.05;
  const colour = faded ? '#c7c3ba' : (pct >= 0 ? 'var(--green)' : 'var(--status-red)');
  const textCol = faded ? 'var(--text-muted)' : (pct >= 0 ? 'var(--green)' : 'var(--status-red)');

  return (
    <div
      onClick={onClick}
      onMouseMove={onHover}
      onMouseLeave={onLeave}
      style={{
        display: 'flex', alignItems: 'center', gap: 11, marginBottom: 12, cursor: 'pointer',
        background: selected ? 'var(--red-tint)' : 'transparent', borderRadius: 8,

function cellColour(cell) {
  if (!cell) return 'var(--surface-sunk)';
  const { pct, p } = cell;
  if (p > 0.05) return '#eae7e0';
  const strong = Math.abs(pct) > 100;
  return pct >= 0 ? (strong ? '#2e9e60' : '#8fd0a8') : (strong ? '#d23a3a' : '#f0a58c');
}

const CATEGORY_ORDER = ['Performance', 'Reward & recognition', 'Tenure & team', 'Demographics'];

function kpiFact(label, market, allMarketKpis) {
  const val = (m, key) => allMarketKpis[m]?.[key] ?? 0;
  const avg = (key) => Object.values(allMarketKpis).reduce((s, k) => s + k[key], 0) / Object.keys(allMarketKpis).length;
  if (label === 'Headcount') {
    const biggest = Object.keys(allMarketKpis).reduce((a, b) => allMarketKpis[a].n > allMarketKpis[b].n ? a : b);
    return market === biggest ? `${market} is the largest population of the three markets.` : `${market} is smaller than ${biggest}, the largest market.`;
  }
  const key = label === 'Talent' ? 'talentRate' : label === 'Accelerated' ? 'acceleratedRate' : 'topRate';
  const here = val(market, key);
  const diff = here - avg(key);
  const dir = diff >= 0 ? 'above' : 'below';
  const rank = Object.keys(allMarketKpis).sort((a, b) => val(b, key) - val(a, key)).indexOf(market) + 1;
  return `At ${here.toFixed(1)}%, ${market} is ${Math.abs(diff).toFixed(1)} pts ${dir} the cross-market average — rank ${rank} of ${Object.keys(allMarketKpis).length}.`;
}

function Kpi({ label, count, pct, accent, fact }) {
  const [flipped, setFlipped] = useState(false);
  return (

          <div style={{ position: 'absolute', left: 0, top: 0, bottom: 0, width: 3, background: accent }} />

        </div>
      </div>
    </div>
  );
}

function buildSummary(markets, rows, block) {
  const goal = block.replace('_vs_Talent', '');
  if (rows.length === 0) return 'No data for this transition.';
  const inAll = [];
  const inOne = [];
  rows.forEach(r => {
    const present = markets.filter(m => r.cells[m] && r.cells[m].p <= 0.05);
    if (present.length === markets.length) inAll.push(r);
    else if (present.length === 1) inOne.push({ r, market: present[0] });
  });
  const strongest = rows[0];
  const strongestMkt = markets.reduce((best, m) => {
    const v = Math.abs(strongest.cells[m]?.pct ?? 0);
    return v > (best.v ?? 0) ? { m, v } : best;
  }, {}).m;
  let s = `Across markets, ${nice(strongest.feature).toLowerCase()} is the standout driver of ${goal}, strongest in ${strongestMkt}.`;
  if (inAll.length > 0 && inAll[0].feature !== strongest.feature) s += ` ${nice(inAll[0].feature)} matters consistently in all three markets.`;
  else if (inAll.length > 0) s += ` It's a reliable signal in all three markets.`;
  if (inOne.length > 0) s += ` ${nice(inOne[0].r.feature)} only shows up in ${inOne[0].market} — a market-specific pattern.`;
  return s;
}

              onMouseEnter={e => { e.currentTarget.style.borderColor = 'var(--gold-soft)'; e.currentTarget.style.color = 'var(--red)'; }}
              onMouseLeave={e => { e.currentTarget.style.borderColor = 'var(--border-strong)'; e.currentTarget.style.color = 'var(--text-secondary)'; }}
              style={{ fontSize: 11, color: 'var(--text-secondary)', background: '#fff', border: '1px solid var(--border-strong)', borderRadius: 15, padding: '6px 11px', cursor: 'pointer', transition: 'all .15s' }}
            >
              {c}
            </div>
          ))}
        </div>
      )}

      <div style={{ marginBottom: 20 }}>
        <p style={rh}>◔ WHY DOES THIS HAPPEN?</p>
        <p style={secText}>{whyText}</p>
      </div>

      <div style={{ marginBottom: 20 }}>
        <p style={rh}>▤ RECOMMENDATIONS</p>
        <p style={secText}>{recText}</p>
      </div>

      <div>
        <p style={rh}>↗ POTENTIAL FOLLOW-UPS</p>
        {followUps.map((f, i) => (
          <div
            key={i}
            onClick={() => ask(f)}
            onMouseEnter={e => { e.currentTarget.style.color = 'var(--red)'; e.currentTarget.style.paddingLeft = '5px'; }}
            onMouseLeave={e => { e.currentTarget.style.color = 'var(--text-secondary)'; e.currentTarget.style.paddingLeft = '0'; }}
            style={{ fontSize: 12, color: 'var(--text-secondary)', padding: '10px 0', borderBottom: '1px solid var(--border)', cursor: 'pointer', transition: 'color .15s, padding-left .15s' }}
          >
            {f}
          </div>
        ))}
      </div>
    </div>
  );
}

import { useState, useMemo } from 'react';
import models from './data/models.json';
import Layout from './components/Layout';
import DeepDive from './views/DeepDive';
import OverviewView from './views/OverviewView';
import DriverView from './views/DriverView';
import HeatmapView from './views/HeatmapView';
import SimulationView from './views/SimulationView';

const MARKETS = Object.keys(models);
const BLOCKS = Object.keys(models[MARKETS[0]].models);

function combinedRateRows(marketData) {
  const cat = marketData.rates['Categorical Base Rates'] ?? [];
  const bin = marketData.rates['Binary Base Rates'] ?? [];
  return [...cat, ...bin];
}

function marketKpi(marketData) {
  const cat = marketData.rates['Categorical Base Rates'] ?? [];
  const bin = marketData.rates['Binary Base Rates'] ?? [];
  const rows = [...cat, ...bin];
  const genderRows = rows.filter(r => r.feature === 'Gender');
  const n = genderRows.reduce((s, r) => s + r.n, 0);
  const talent = genderRows.reduce((s, r) => s + r.talent, 0);
  const accelerated = genderRows.reduce((s, r) => s + r.accelerated, 0);
  const top = genderRows.reduce((s, r) => s + r.top, 0);
  return {
    n,
    talentRate: n ? (talent / n) * 100 : 0,
    acceleratedRate: n ? (accelerated / n) * 100 : 0,
    topRate: n ? (top / n) * 100 : 0,
  };
}

export default function App() {
  const [active, setActive] = useState('overview');
  const [market, setMarket] = useState(MARKETS[0]);
  const [block, setBlock] = useState(BLOCKS[0]);
  const [feature, setFeature] = useState(null);
  const [group, setGroup] = useState(null);

  const marketData = models[market];
  const rateRows = useMemo(() => combinedRateRows(marketData), [marketData]);

  const featureOptions = useMemo(
    () => [...new Set(rateRows.map(r => r.feature))],
    [rateRows]
  );
  const groupOptions = useMemo(
    () => feature ? [...new Set(rateRows.filter(r => r.feature === feature).map(r => r.group))] : [],
    [rateRows, feature]
  );

  const driverRows = marketData.models[block];

  const kpi = useMemo(() => {
    if (feature && group) {
      const row = rateRows.find(r => r.feature === feature && r.group === group);
      if (row) return { n: row.n, talentRate: row.talentRate, acceleratedRate: row.acceleratedRate, topRate: row.topRate };
    }
    const genderRows = rateRows.filter(r => r.feature === 'Gender');
    const n = genderRows.reduce((s, r) => s + r.n, 0);
    const talent = genderRows.reduce((s, r) => s + r.talent, 0);
    const accelerated = genderRows.reduce((s, r) => s + r.accelerated, 0);
    const top = genderRows.reduce((s, r) => s + r.top, 0);
    return {
      n,
      talentRate: n ? (talent / n) * 100 : 0,
      acceleratedRate: n ? (accelerated / n) * 100 : 0,
      topRate: n ? (top / n) * 100 : 0,
    };
  }, [rateRows, feature, group]);

  // cross-market KPI stats for comparison facts
  const allMarketKpis = useMemo(() => {
    const out = {};
    MARKETS.forEach(m => { out[m] = marketKpi(models[m]); });
    return out;
  }, []);

  const views = {
    overview: <OverviewView driverRows={driverRows} market={market} block={block} />,
    driver: <DriverView kpi={kpi} driverRows={driverRows} block={block} market={market} allMarketKpis={allMarketKpis} />,
    heatmap: <HeatmapView kpi={kpi} models={models} block={block} market={market} allMarketKpis={allMarketKpis} />,
    simulation: <SimulationView models={models} market={market} block={block} />,
  };

  const deepDive = <DeepDive driverRows={driverRows} market={market} block={block} />;

  return (
    <Layout
      active={active}
      onNavigate={setActive}
      markets={MARKETS}
      market={market}
      setMarket={setMarket}
      blocks={BLOCKS}
      block={block}
      setBlock={setBlock}
      featureOptions={featureOptions}
      feature={feature}
      setFeature={setFeature}
      groupOptions={groupOptions}
      group={group}
      setGroup={setGroup}
      deepDive={deepDive}
    >
      {views[active]}
    </Layout>
  );
}
